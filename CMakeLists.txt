# ---------------------------------------------------------------------------
# Capabilities
#
# MT_CAPS_FRAGMENT is written by tools/mtcaps into $MT_OUT -- outside every
# checkout -- and passed in by build-linux.sh. When it is absent this is a
# STANDALONE build and the option() defaults above apply, which is what "clone
# the engine and build it" must keep doing.
#
# The fragment sets each MT_ENABLE_* to 0 or 1 as ordinary variables, so it must
# be included AFTER the option() calls and BEFORE anything consults them.
# ---------------------------------------------------------------------------
if (DEFINED MT_CAPS_FRAGMENT)
    if (NOT EXISTS "${MT_CAPS_FRAGMENT}")
        message(FATAL_ERROR
            "MT_CAPS_FRAGMENT=${MT_CAPS_FRAGMENT} does not exist.\n"
            "It is generated by tools/mtcaps. Build through the wrapper:\n"
            "  bash ${CMAKE_CURRENT_SOURCE_DIR}/build-linux.sh --manifest <app>/mtengine.caps --app <app>")
    endif()
    include("${MT_CAPS_FRAGMENT}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MT_CAPS_FRAGMENT}")
    message(STATUS "MTEngineSDL capabilities: ${MT_CAPS_RESOLVED}")
else()
    message(STATUS "MTEngineSDL: STANDALONE build -- engine option() defaults, no manifest")

    # No fragment, so nothing put the generated include dir on the path -- but
    # llama_cpp_version.h lives there now (it used to be a TRACKED file that
    # build-llama_cpp.sh rewrote on every build) and
    # CGuiViewLlamaModelLoader.cpp includes it unconditionally.
    if (DEFINED ENV{MTENGINE_BUILD_ROOT})
        set(MT_STANDALONE_INCLUDE "$ENV{MTENGINE_BUILD_ROOT}/_standalone/include")
    elseif (DEFINED ENV{XDG_CACHE_HOME})
        set(MT_STANDALONE_INCLUDE "$ENV{XDG_CACHE_HOME}/mtengine/_standalone/include")
    else()
        set(MT_STANDALONE_INCLUDE "$ENV{HOME}/.cache/mtengine/_standalone/include")
    endif()
    include_directories("${MT_STANDALONE_INCLUDE}")
endif()

# Disable features if submodules are still missing after auto-checkout attempt
# A capability that was ASKED FOR and whose inputs are absent is a hard error
# naming the command that fixes it. There is no downgrade tier.
#
# What these three used to do was warn and turn themselves off, which sounds kind
# and behaves badly: a fresh checkout did not fail, it silently produced a binary
# without the feature and the user found out at runtime. A build that stops with
# "run this command" is strictly better than one that succeeds and lies about
# what it is.

# PLATFORM defines only. MT_ENABLE_IMGUI_TEST_ENGINE and MT_CAMERA_CAPTURE_ENABLED
# used to be here and could not stay, because add_definitions() is DIRECTORY
# scope and CMake emits directory definitions AFTER a target's own:
#
#     add_definitions(-DFOO=1) + target_compile_definitions(t PUBLIC FOO=0)
#       ->  C_DEFINES = -DFOO=0 -DFOO=1        (measured, cmake 3.31)
#
# So the hard-wired value won the preprocessor and MT_CAP_TEST_ENGINE=0 /
# MT_CAP_CAMERA=0 shipped a Linux binary with the feature compiled IN -- with a
# -Wmacro-redefined per TU as the only hint. They are capability-owned now and
# are emitted as TARGET definitions below, where the fragment can win.
add_definitions(
    -DLINUX
    -DIMGUI_IMPL_OPENGL_LOADER_GL3W
)

# THE ONE SCHEDULED EXCEPTION to the hard-error rule, and it is bounded.
#
# libheif is a SYSTEM package on Linux and nothing in this tree installs it, so a
# hard error here would make every Linux machine without libheif-dev unable to
# build any app whose manifest asks for photo codecs. It converts to FATAL_ERROR
# in the same commit that VENDORS libheif -- so the error can only ever fire for
# a cause the build itself can fix. Do not add a second exception without writing
# its expiry into a phase.
if(MT_ENABLE_LIBHEIF AND NOT APPLE)
    pkg_check_modules(HEIF libheif)
    if(NOT HEIF_FOUND)
        if(DEFINED MT_CAPS_FRAGMENT)
            # THE EXCEPTION DOES NOT SURVIVE A MANIFEST-DRIVEN BUILD, and this is
            # the honest scope of it rather than a second exception.
            #
            # This process can downgrade its OWN MT_ENABLE_LIBHEIF -- and now
            # does, because mt_apply_capabilities re-reads the variable. But the
            # APP is configured by a SEPARATE cmake process that includes the
            # same fragment and never runs this pkg_check_modules, so the app's
            # TUs would keep MT_ENABLE_LIBHEIF=1 while the engine's had 0.
            # the photo app guards on that flag in its own src/, so the two sides
            # would disagree about a feature -- the ODR shape, one flag over.
            #
            # A silent divergence is worse than a stopped build, so under a
            # manifest this is an error naming both ways out.
            message(FATAL_ERROR
                "MT_CAP_PHOTO_CODECS is on but libheif was not found.\n"
                "Install it:            sudo apt-get install -y libheif-dev\n"
                "Or turn it off:        MT_CAP_PHOTO_CODECS=0 in the app's mtengine.caps\n"
                "(The warn-and-downgrade path is kept only for STANDALONE engine "
                "builds, where there is no app process to disagree with.)")
        endif()
        message(WARNING "libheif not found — HEIC/HEIF disabled on this STANDALONE build. Install: apt install libheif-dev")
        set(MT_ENABLE_LIBHEIF OFF CACHE BOOL "" FORCE)
    endif()
endif()

# Apply the resolved capability set.
#
# PUBLIC, so it reaches the app's translation units too and not only the
# engine's. That is not tidiness: MT_ENABLE_MBEDTLS appears in httplib.h and
# changes httplib::Client's struct LAYOUT, so an engine built with it on and an
# app TU compiled with it off allocate different-sized objects. PRIVATE is how
# that bug happened.
#
# This also puts $MT_OUT/include on the include path, where MT_Capabilities.h
# lives, before anything compiles.
# ---------------------------------------------------------------------------
if (DEFINED MT_CAPS_FRAGMENT)
    mt_apply_capabilities(MTEngineSDL)
endif()

# The two that moved out of add_definitions(). Emitted with an EXPLICIT value
# either way: emitting only the on ones would leave the off ones UNDEFINED, and
# the #ifdef family defaults off while the value style defaults on -- so "absent"
# means two different things depending on the guard it meets.
#
# When a fragment is in play these repeat what mt_apply_capabilities already
# added, at the same value, which is an identical redefinition and therefore
# silent. When there is none they are the standalone defaults.
target_compile_definitions(MTEngineSDL PUBLIC
    MT_ENABLE_IMGUI_TEST_ENGINE=$<BOOL:${MT_ENABLE_IMGUI_TEST_ENGINE}>
    MT_CAMERA_CAPTURE_ENABLED=$<BOOL:${MT_CAMERA_CAPTURE_ENABLED}>
)

# Camera on Linux: NO libv4l2 LINK, and that is a measurement rather than an
# omission.
#
# The engine used to `target_link_libraries(MTEngineSDL PRIVATE v4l2)`
# unconditionally, and I briefly made that a FATAL_ERROR when MT_CAP_CAMERA was
# on -- which would have required libv4l-dev on every Linux box building DummyApp
# or c64d. Measured across the whole engine:
#
#     libv4l2 API calls (v4l2_open/v4l2_ioctl/v4l2_mmap/...) : 0
#     #include <libv4l2.h>                                    : 0
#
# platform/Linux/src.Linux/CCameraLinux.cpp uses the KERNEL UAPI header
# <linux/videodev2.h> -- which ships with linux-libc-dev and is always present --
# with plain open/ioctl/mmap. It never touches the libv4l2 shim. So the link was
# an over-link from the start, and making it mandatory would have been a new
# hard dependency on a library nothing calls.
#
# The capability is gated where it belongs: Core/CCameraCapture.cpp guards the
# platform factory on MT_CAMERA_CAPTURE_ENABLED and falls back to CCameraDummy.
# CCameraLinux.cpp stays in the file list and compiles harmlessly either way --
# its class is simply never instantiated -- which is the guard-the-body,
# keep-the-file idiom the rest of the programme uses.

# Image codec bundle (TIFF, WebP, AVIF, LibRaw)
if(EXISTS "${MT_IMAGE_CODECS_LIB}")
    if(MT_ENABLE_LIBTIFF)
        target_compile_definitions(MTEngineSDL PUBLIC MT_ENABLE_LIBTIFF=1)
    endif()
    if(MT_ENABLE_LIBWEBP)
        target_compile_definitions(MTEngineSDL PUBLIC MT_ENABLE_LIBWEBP=1)
    endif()
    if(MT_ENABLE_LIBAVIF)
        target_compile_definitions(MTEngineSDL PUBLIC MT_ENABLE_LIBAVIF=1)
    endif()
    if(MT_ENABLE_LIBRAW)
        target_compile_definitions(MTEngineSDL PUBLIC MT_ENABLE_LIBRAW=1)
    endif()
    if(EXISTS "${MT_IMAGE_CODECS_INCLUDE}")
        target_include_directories(MTEngineSDL PRIVATE "${MT_IMAGE_CODECS_INCLUDE}")
    endif()
    # lcms2 backs MT_CAP_COLOR_MANAGEMENT. It is gated on the OPTION now, not on
    # "does lcms2.h happen to exist": the filesystem no longer votes.
    #
    # What it used to do is worth recording, because it was the worst of the
    # seven degradation sites. It had no option() at all, so unlike every other
    # flag it could not even be overridden with -D; and its else() branch sat
    # INSIDE if(EXISTS "${MT_IMAGE_CODECS_LIB}"), so when the bundle was missing
    # altogether the warning never printed and colour management silently became
    # an identity transform with no diagnostic whatsoever. A manifest could ask
    # for colour management and get nothing, with no signal.
    if(MT_ENABLE_LCMS2)
        if(NOT EXISTS "${MT_IMAGE_CODECS_INCLUDE}/lcms2.h")
            message(FATAL_ERROR
                "MT_CAP_COLOR_MANAGEMENT is on but lcms2.h is not in the image-codecs bundle.\n"
                "Run: bash ${CMAKE_CURRENT_SOURCE_DIR}/platform/Linux/build-image_codecs.sh")
        endif()
        target_compile_definitions(MTEngineSDL PUBLIC MT_ENABLE_LCMS2=1)
    else()
        target_compile_definitions(MTEngineSDL PUBLIC MT_ENABLE_LCMS2=0)
    endif()
    target_link_libraries(MTEngineSDL PRIVATE "${MT_IMAGE_CODECS_LIB}")
endif()
